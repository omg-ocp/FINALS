                // Normalize to shortest path around circle
                if (relativeIndex > totalChars / 2) relativeIndex -= totalChars;
                if (relativeIndex < -totalChars / 2) relativeIndex += totalChars;
                
                // Calculate 3D position
                const angle = relativeIndex * angleStep;
                const radian = (angle * Math.PI) / 180;
                
                const x = Math.sin(radian) * radius;
                const z = Math.cos(radian) * radius - radius;
                const rotateY = -angle;
                
                // Scale based on z position (closer = bigger)
                const scale = 0.6 + (z + radius) / (radius * 2) * 0.6;
                const opacity = 0.3 + (z + radius) / (radius * 2) * 0.7;

        // ===== INFINITY STONE CURSOR =====
        (function() {
            console.log('Cursor script starting...');
            
            // Create cursor element
            const cursor = document.createElement('div');
            cursor.className = 'infinity-cursor';
            document.body.appendChild(cursor);
            console.log('Cursor element created:', cursor);

            // Stone types for each section
            const sectionStones = {
                'hero-section': 'power',
                'home': 'space',
                'about': 'mind',
                'cast': 'reality',
                'gallery': 'time',
                'reviews': 'soul'
            };

            let currentStone = 'power';
            let mouseX = 0, mouseY = 0;
            let cursorX = 0, cursorY = 0;
            let particles = [];
            let particleInterval = null;

            // Smooth cursor movement
            function animateCursor() {
                const dx = mouseX - cursorX;
                const dy = mouseY - cursorY;
                cursorX += dx * 0.35;
                cursorY += dy * 0.35;
                cursor.style.left = cursorX + 'px';
                cursor.style.top = cursorY + 'px';
                requestAnimationFrame(animateCursor);
            }
            animateCursor();

            // Track mouse position
            document.addEventListener('mousemove', (e) => {
                mouseX = e.clientX;
                mouseY = e.clientY;
                
                // Detect which section we're in
                const sections = document.querySelectorAll('section');
                sections.forEach(section => {
                    const rect = section.getBoundingClientRect();
                    if (e.clientY >= rect.top && e.clientY <= rect.bottom) {
                        let sectionId = section.id || '';
                        if (section.classList.contains('hero-section')) {
                            sectionId = 'hero-section';
                        }
                        const stone = sectionStones[sectionId];
                        if (stone && stone !== currentStone) {
                            currentStone = stone;
                            updateStoneColor(stone);
                        }
                    }
                });
            });

            // Update stone color
            function updateStoneColor(stone) {
                cursor.className = 'infinity-cursor';
                if (stone !== 'power') {
                    cursor.classList.add('stone-' + stone);
                }
            }

            // Create particle trail
            function createParticle(x, y) {
                const particle = document.createElement('div');
                particle.className = 'cursor-particle ' + currentStone;
                particle.style.left = (x + (Math.random() - 0.5) * 10) + 'px';
                particle.style.top = (y + (Math.random() - 0.5) * 10) + 'px';
                document.body.appendChild(particle);
                
                setTimeout(() => {
                    particle.remove();
                }, 600);
            }

            document.addEventListener('mousemove', (e) => {
                const now = Date.now();

                    createParticle(e.clientX, e.clientY);

                }

        // ===== SECTION MUSIC SYSTEM =====
        (function() {
            // Music configuration per section
            const sectionMusic = {
                'hero-section': 'mp3 src/Whatever it takes.mp3',
                'home': 'mp3 src/Portals.mp3',
                'about': 'mp3 src/The real hero.mp3',
                'cast': 'mp3 src/Five Seconds.mp3',
                'gallery': 'mp3 src/Main on end.mp3',
                'reviews': 'mp3 src/The real hero (2).mp3'
            };

            // Create audio elements for each section
            const audioElements = {};
            let currentSection = null;
            let isMuted = false;
            let musicStarted = false;
            const fadeTime = 800; // ms for fade in/out

            // Create all audio elements
            Object.keys(sectionMusic).forEach(section => {
                const audio = new Audio(sectionMusic[section]);
                audio.loop = true;
                audio.volume = 0;
                audio.preload = 'auto';
                audioElements[section] = audio;
            });

            // Fade functions
            function fadeIn(audio, duration = fadeTime) {
                if (!audio || isMuted) return;
                audio.volume = 0;
                audio.play().catch(e => {});
                
                const steps = 20;
                const stepTime = duration / steps;
                const volumeStep = 0.3 / steps; // Max volume 0.3
                let currentStep = 0;
                
                const fadeInterval = setInterval(() => {
                    currentStep++;
                    if (currentStep >= steps || isMuted) {
                        audio.volume = isMuted ? 0 : 0.3;
                        clearInterval(fadeInterval);
                    } else {
                        audio.volume = Math.min(0.3, volumeStep * currentStep);
                    }
                }, stepTime);
            }

            function fadeOut(audio, duration = fadeTime) {
                if (!audio) return;
                
                const startVolume = audio.volume;
                const steps = 20;
                const stepTime = duration / steps;
                const volumeStep = startVolume / steps;
                let currentStep = 0;
                
                const fadeInterval = setInterval(() => {
                    currentStep++;
                    if (currentStep >= steps) {
                        audio.volume = 0;
                        audio.pause();
                        audio.currentTime = 0;
                        clearInterval(fadeInterval);
                    } else {
                        audio.volume = Math.max(0, startVolume - (volumeStep * currentStep));
                    }
                }, stepTime);
            }

            // Switch music between sections
            function switchMusic(newSection) {
                if (newSection === currentSection || !musicStarted) return;
                
                // Fade out current
                if (currentSection && audioElements[currentSection]) {
                    fadeOut(audioElements[currentSection]);
                }
                
                // Fade in new
                if (audioElements[newSection]) {
                    setTimeout(() => {
                        fadeIn(audioElements[newSection]);
                    }, fadeTime / 2);
                }
                
                currentSection = newSection;
            }

            // Detect current section based on scroll
            function detectSection() {
                const scrollPosition = window.scrollY + window.innerHeight / 2;
                
                // Check hero section first
                const heroSection = document.querySelector('.hero-section');
                if (heroSection) {
                    const rect = heroSection.getBoundingClientRect();
                    if (rect.top <= window.innerHeight / 2 && rect.bottom > window.innerHeight / 2) {
                        return 'hero-section';
                    }
                }
                
                // Check other sections
                const sections = document.querySelectorAll('section[id]');
                let detected = 'hero-section';
                
                sections.forEach(section => {
                    const sectionTop = section.offsetTop;
                    const sectionHeight = section.offsetHeight;
                    const sectionId = section.getAttribute('id');
                    
                    if (scrollPosition >= sectionTop && scrollPosition < sectionTop + sectionHeight) {
                        detected = sectionId;
                    }
                });
                
                return detected;
            }

            // Listen for scroll to switch music
            window.addEventListener('scroll', () => {
                if (!musicStarted) {
                    initMusic();
                }
                const section = detectSection();
                switchMusic(section);
            });

            // Start music on first user interaction OR immediately on page load
            function initMusic() {
                if (musicStarted) return;
                musicStarted = true;
                
                const section = detectSection();
                currentSection = section;
                
                if (audioElements[section] && !isMuted) {
                    fadeIn(audioElements[section]);
                }
            }

            // Auto-start music on page load
            window.addEventListener('load', () => {
                setTimeout(() => {
                    initMusic();
                }, 500); // Small delay to ensure everything is ready
            });

            // Also start on any interaction as fallback
            document.addEventListener('click', initMusic, { once: true });
            document.addEventListener('touchstart', initMusic, { once: true });
            document.addEventListener('mousemove', initMusic, { once: true });

            // Create sound toggle button
            const toggleBtn = document.createElement('button');
            toggleBtn.id = 'sound-toggle';
            toggleBtn.innerHTML = '🔊';
            toggleBtn.title = 'Toggle Music';
            toggleBtn.style.cssText = `
                position: fixed;
                bottom: 20px;
                right: 20px;
                width: 50px;
                height: 50px;
                border-radius: 50%;
                border: 2px solid rgba(138, 43, 226, 0.6);
                background: rgba(0, 0, 0, 0.7);
                color: white;
                font-size: 24px;
                cursor: pointer;
                z-index: 99998;
                transition: all 0.3s ease;
                display: flex;
                align-items: center;
                justify-content: center;
                backdrop-filter: blur(5px);
            `;
            
            toggleBtn.addEventListener('mouseenter', () => {
                toggleBtn.style.transform = 'scale(1.1)';
                toggleBtn.style.borderColor = 'rgba(138, 43, 226, 1)';
                toggleBtn.style.boxShadow = '0 0 15px rgba(138, 43, 226, 0.5)';
            });
            
            toggleBtn.addEventListener('mouseleave', () => {
                toggleBtn.style.transform = 'scale(1)';
                toggleBtn.style.borderColor = 'rgba(138, 43, 226, 0.6)';
                toggleBtn.style.boxShadow = 'none';
            });

            toggleBtn.addEventListener('click', (e) => {
                e.stopPropagation();
                isMuted = !isMuted;
                toggleBtn.innerHTML = isMuted ? '🔇' : '🔊';
                
                if (isMuted) {
                    // Mute all
                    Object.values(audioElements).forEach(audio => {
                        audio.volume = 0;
                        audio.pause();
                    });
                } else {
                    // Resume current section music
                    if (musicStarted && currentSection && audioElements[currentSection]) {
                        fadeIn(audioElements[currentSection]);
                    }
                }
            });

            document.body.appendChild(toggleBtn);

            // Handle nav-link clicks to switch music immediately
            document.querySelectorAll('.nav-link').forEach(link => {
                link.addEventListener('click', function() {
                    if (!musicStarted) {
                        initMusic();
                    }
                    const targetId = this.getAttribute('href').substring(1);
                    if (audioElements[targetId]) {
                        setTimeout(() => switchMusic(targetId), 100);
                    }
                });
            });

            // Handle scroll-top clicks (restart hero music)
            document.querySelectorAll('.scroll-top').forEach(link => {
                link.addEventListener('click', function() {
                    if (!musicStarted) {
                        initMusic();
                    }
                    setTimeout(() => {
                        // Restart hero section music
                        if (audioElements['hero-section']) {
                            fadeOut(audioElements[currentSection]);
                            currentSection = 'hero-section';
                            audioElements['hero-section'].currentTime = 0;
                            setTimeout(() => fadeIn(audioElements['hero-section']), fadeTime / 2);
                        }
                    }, 100);
                });
            });
        })();
